// Diagnostic experiment, not a business acceptance test. All data and HTTP responses are synthetic.
var at = DateTimeOffset.Parse("2026-10-18T02:00:00Z");
foreach (var scenario in new[] { "resolved", "device-missing", "device-not-found", "site-timezone-missing", "shift-not-found", "legacy" })
{
    using var http = new FixtureHandler(scenario);
    ProductionReportOeeDimensionSnapshot? snapshot = scenario == "legacy" ? null :
        await new HttpProductionReportOeeDimensionSnapshotProvider(
            new MesMasterDataHttpClient(new HttpClient(http) { BaseAddress = new Uri("http://fixture.invalid") }),
            new SyntheticToken()).CaptureAsync(new("probe-org", "probe-env",
                scenario == "device-missing" ? null : "PROBE-DEVICE", "PROBE-WC", "DAY"), CancellationToken.None);
    await using var db = new ApplicationDbContext(new DbContextOptionsBuilder<ApplicationDbContext>()
        .UseInMemoryDatabase($"probe-{Guid.NewGuid():N}").Options, new NoopMediator());
    db.WorkOrders.Add(WorkOrder.Create("probe-org", "probe-env", "PROBE-WO", "PROBE-SKU", "PROBE-PV", 2m, 10, at.AddDays(1), "PCS"));
    var report = ProductionReport.Record("probe-org", "probe-env", "PROBE-REPORT", "PROBE-WO", "PROBE-OP", 2m, 0m, false, at,
        oeeProjection: new("PROBE-WC", "PROBE-DEVICE", "PCS", 10m), oeeDimensionSnapshot: snapshot);
    var payload = new ProductionReportRecordedIntegrationEventConverter().Convert(
        report.GetDomainEvents().OfType<ProductionReportRecordedDomainEvent>().Single()).Payload;
    db.ProductionReports.Add(report);
    await db.SaveChangesAsync();
    db.ChangeTracker.Clear();
    var saved = await db.ProductionReports.AsNoTracking().SingleAsync();
    var buckets = new Dictionary<string, object>();
    foreach (var dimension in new[] { ProductionStatisticsDimension.Day, ProductionStatisticsDimension.WorkCenter, ProductionStatisticsDimension.Sku })
    {
        var result = await new QueryProductionStatisticsQueryHandler(db).Handle(new("probe-org", "probe-env", dimension, at.AddMinutes(-30), at.AddMinutes(30)), CancellationToken.None);
        buckets[dimension.ToString()] = result.Items;
    }
    Console.WriteLine(JsonSerializer.Serialize(new { scenario, requests = http.Requests,
        saved.OeeDimensionResolutionStatus, saved.OeeDimensionDegradedReason, saved.OeeSiteCode, saved.OeeSiteTimezone,
        eventSiteCode = payload.SiteCode, eventTimezone = payload.SiteTimezone, buckets }));
}
sealed class FixtureHandler(string scenario) : HttpMessageHandler
{
    public List<string> Requests { get; } = [];
    protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken cancellationToken)
    {
        Requests.Add(request.RequestUri!.PathAndQuery);
        object[] resources;
        if (request.RequestUri.Query.Contains("resourceType=device-asset"))
            resources = scenario == "device-not-found" ? [] : [new { resourceType = "device-asset", code = "PROBE-DEVICE", displayName = "probe", active = true, snapshotVersion = "v1", deviceAssetId = "019c9c62-9987-7af2-8fa2-3fd936098265", workCenterCode = "PROBE-WC", siteCode = "PROBE-SITE" }];
        else if (request.RequestUri.Query.Contains("resourceType=site"))
            resources = [new { resourceType = "site", code = "PROBE-SITE", displayName = "probe", active = true, snapshotVersion = "v1", timezone = scenario == "site-timezone-missing" ? null : "Asia/Shanghai" }];
        else resources = scenario == "shift-not-found" ? [] : [new { resourceType = "shift", code = "DAY", displayName = "probe", active = true, snapshotVersion = "v1", startsAt = "08:00:00", endsAt = "16:00:00", crossesMidnight = false, paidMinutes = 450, breakMinutes = 30 }];
        return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK) { Content = new StringContent(JsonSerializer.Serialize(new { data = new { resources, total = resources.Length, truncated = false } }), Encoding.UTF8, "application/json") });
    }
}
sealed record SyntheticToken : IInternalServiceTokenProvider { public string BearerToken => "synthetic-probe-token"; }
sealed class NoopMediator : IMediator
{
    public Task Publish(object notification, CancellationToken cancellationToken = default) => Task.CompletedTask;
    public Task Publish<T>(T notification, CancellationToken cancellationToken = default) where T : INotification => Task.CompletedTask;
    public Task<T> Send<T>(IRequest<T> request, CancellationToken cancellationToken = default) => throw new NotSupportedException();
    public Task Send<T>(T request, CancellationToken cancellationToken = default) where T : IRequest => throw new NotSupportedException();
    public Task<object?> Send(object request, CancellationToken cancellationToken = default) => throw new NotSupportedException();
    public IAsyncEnumerable<T> CreateStream<T>(IStreamRequest<T> request, CancellationToken cancellationToken = default) => throw new NotSupportedException();
    public IAsyncEnumerable<object?> CreateStream(object request, CancellationToken cancellationToken = default) => throw new NotSupportedException();
}
